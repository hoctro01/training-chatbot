🤖 Phần 1: Bot là gì?

Bot = chương trình tự động thực hiện tác vụ thay con người, chạy theo lịch hoặc phản hồi khi được gọi.

Loại Đặc điểm Ví dụ
Notification Gửi thông báo 1 chiều theo lịch Daily report, alert
Command Nhận lệnh → trả kết quả /status, /report
Chatbot Hội thoại rule-based FAQ bot
AI Chatbot Hội thoại tự nhiên qua LLM BOT KSVHKT ✨
💡 BOT KSVHKT = Notification + Command + AI Chatbot — kết hợp cả 3 loại!

🚀 Phần 5b: main.py — Entry point (code thật)


# src/automation_bot/main.py
import argparse, sys
from automation_bot.job import run_automation

def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="BOT KSVHKT")
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--once",  action="store_true", help="Chạy 1 lần → exit")
    group.add_argument("--serve", action="store_true", help="Chatbot WebSocket")
    parser.add_argument("--dry-run", action="store_true", help="Không gửi NetChat")
    args = parser.parse_args(argv)

    if args.once:
        result = run_automation(dry_run=args.dry_run)
        if not result.ok:
            print(f"Failed: {result.summary}", file=sys.stderr)
            return 3                            # ← exit code != 0 → lỗi
        print("Dry-run OK" if args.dry_run else "Sent OK")
        return 0                                # ← exit code 0 → OK

    if args.serve:
        from automation_bot.websocket_listener import start_listener
        start_listener()                        # ← blocking, chạy mãi mãi
        return 0

if __name__ == "__main__":
    raise SystemExit(main())
📌 Pattern: --once = 1 lần → scheduler gọi lại | --serve = chạy liên tục (daemon/systemd)

# src/automation_bot/job.py — Pattern chuẩn
from dataclasses import dataclass

@dataclass(frozen=True)
class JobResult:
    ok: bool
    summary: str

def run_automation(*, dry_run: bool = False) -> JobResult:
    try:
        # ─── BƯỚC 1: Lấy data ───
        df = get_data_from_source()           # Tableau / API / DB / Excel

        # ─── BƯỚC 2: Xử lý ───
        df = normalize_data(df)               # Clean, convert types
        summary_df = build_report(df)          # Group by, tính KPI

        # ─── BƯỚC 3: Xuất Excel ───
        export_excel(summary_df, file_path)   # openpyxl, có format

        # ─── BƯỚC 4: Tạo message ───
        message = compose_message(summary_df) # Markdown + emoji 🔴🟢

        # ─── BƯỚC 5: Gửi NetChat ───
        if not dry_run:
            send_report_to_netchat(base_url, token, channel_id,
                                   message, file_path)

        return JobResult(ok=True, summary="OK")
    except Exception as exc:
        return JobResult(ok=False, summary=str(exc))

🔌 Phần 5d: WebSocket Listener (code thật)


# websocket_listener.py — Các bước quan trọng

# 1) Auto-detect bot user ID (để bỏ qua message của chính bot)
def _fetch_bot_user_id() -> str:
    resp = requests.get(f"{NET_CHAT_URL}/api/v4/users/me",
                        headers={"Authorization": f"Bearer {BOT_TOKEN}"})
    return resp.json().get("id", "")

# 2) Filter: chỉ xử lý message hợp lệ
def _should_process(event):
    if event.get("event") != "posted": return False
    post = json.loads(event["data"]["post"])
    if post["user_id"] == _bot_user_id: return False  # Bỏ qua message mình
    return True

# 3) Listen loop (async, auto-reconnect)
async def _listen(ws_url):
    while True:
        try:
            async with websockets.connect(ws_url, ...) as ws:
                await ws.send(json.dumps({           # Authenticate
                    "action": "authentication_challenge",
                    "data": {"token": BOT_TOKEN}
                }))
                async for msg in ws:                 # Loop lắng nghe
                    event = json.loads(msg)
                    if _should_process(event) and _is_directed_at_bot(event):
                        loop.run_in_executor(None, _handle_posted_event, event)
        except (ConnectionClosed, OSError):
            await asyncio.sleep(delay)               # Auto reconnect!
            delay = min(delay * 2, 60)               # Exponential backoff

# message_router.py

COMMANDS = {
    "/help":     "help",
    "/report":   "report",
    "/check":    "check",
    "/overdue":  "overdue",
    "/upcoming": "upcoming",
    "/stats":    "stats",
}

def route_message(raw_text, channel_id):
    text = _strip_bot_mention(raw_text)
    command, args = _parse_command(text)

    if command == "help":
        return handle_help()
    elif command == "report":
        return handle_report()
    elif command == "check":
        return handle_check(args)
    elif command == "overdue":
        return handle_overdue()
    else:
        # → Hỏi tự nhiên → LLM
        return handle_question(text, channel_id)

📝 Phần 6b: System Prompt (code thật)


# llm/chatbot_prompt.py

CHATBOT_SYSTEM_PROMPT = """Bạn là trợ lý AI của phòng Kiểm soát vận hành
khai thác (TTKTKV3), chuyên trả lời câu hỏi về dữ liệu SR đúng hạn.

Quy tắc bắt buộc:
- CHỈ dùng dữ liệu được cung cấp. KHÔNG bịa số liệu.
- Nếu thiếu dữ liệu → nói rõ "Tôi không có thông tin này".
- Tiếng Việt, ngắn gọn, chuyên nghiệp.
- Emoji: ✅ đạt, 🔴 không đạt, ⚠️ cảnh báo, 📊 số liệu.
- Format Markdown. Giới hạn 5-10 dòng.
"""

def build_chatbot_messages(question, data_context, history=None):
    messages = [
        {"role": "system",    "content": CHATBOT_SYSTEM_PROMPT},
        {"role": "user",      "content": f"Dữ liệu SR:\\n{data_context}"},
        {"role": "assistant", "content": "Đã nhận dữ liệu. Sẵn sàng trả lời."},
    ]
    if history:                            # Thêm lịch sử hội thoại
        messages.extend(history[-4:])
    messages.append({"role": "user", "content": question})
    return messages
⚠️ QUAN TRỌNG: Luôn "grounding" LLM bằng data thật. Nếu không có data context, LLM sẽ bịa số!

🔗 Phần 6c: call_llm_v2.py — Gọi LLM API


# llm/call_llm_v2.py — Các tính năng quan trọng

def call_llm_chat(messages, *, model=None, token=None, url=None,
                   max_tokens=256, temperature=0.5, timeout=60):
    """Gọi /v1/chat/completions, trả về nội dung assistant."""

    payload = {
        "model": model or "Qwen/Qwen3-32B",
        "messages": messages,
        "max_tokens": max_tokens,
        "temperature": temperature,       # 0=chính xác, 1=sáng tạo
    }

    # Auto retry: 429 (rate limit), 500, 502, 503, 504
    resp = _post_with_retries(url, headers=..., payload=payload, timeout=60)

    content = _strip_think(                # Loại bỏ <think> block
        _extract_chat_content(resp.json()) # Trích content từ response
    )
    return content

✅ Tính năng có sẵn

  • Auto retry + exponential backoff
  • Strip <think> blocks
  • Fallback model name
  • Configurable via .env

🔧 Tuỳ chỉnh khi tạo bot mới

  • Thay model nếu dùng model khác
  • temperature: 0.3-0.5 cho báo cáo
  • max_tokens: tăng nếu cần dài hơn

class TestKPICalculation:

        """Dưới target → Không đạt"""
        result = calc_kpi(on_time=5,

📋 Đề bài: Tạo Bot Giám sát Sự cố mạng

Bot hằng ngày đọc file incidents.xlsx, tính tổng sự cố / đã xử lý / đang mở, gửi báo cáo lên NetChat + hỗ trợ hỏi đáp AI.

Yêu cầu

  1. Đọc incidents.xlsx (cột: id, type, status, department, date)
  2. Tính KPI: total, resolved, open, MTTR
  3. Gửi báo cáo Markdown lên NetChat
  4. Command: /incidents, /critical
  5. AI trả lời: "Sự cố nào nghiêm trọng nhất?"

Hướng dẫn từng bước

  1. Copy chatbot_ksvhkt → chatbot_incident
  2. Sửa process_data/ → đọc Excel thay Tableau
  3. Sửa job.py → logic tính sự cố + format message
  4. Thêm /incidents, /critical vào router
  5. Viết system prompt cho domain "sự cố mạng"
  6. Test: --dry-run → --once → --serve
